const roomId = 1;

const socket = new WebSocket(`ws://localhost:8000/ws/chat?token=${token}&room_id=${roomId}`);

socket.onopen = () => {
  console.log("✅ WebSocket 연결됨");

  // 메시지 전송
  socket.send(JSON.stringify({
    type: "message",
    room_id: roomId,
    message: "프론트에서 보낸 테스트 메시지입니다."
  }));

  // 읽음 처리
  setTimeout(() => {
    socket.send(JSON.stringify({
      type: "read",
      message_id: 1  // ✅ 실제 존재하는 메시지 ID
    }));
  }, 1000);

  // 접속 중 유저 조회
  setTimeout(() => {
    socket.send(JSON.stringify({
      type: "online_users"
    }));
  }, 1500);

  // 이전 메시지 조회
  setTimeout(() => {
    socket.send(JSON.stringify({
      type: "fetch_old_messages",
      before: new Date().toISOString(),
      limit: 10
    }));
  }, 2000);
};

socket.onmessage = (event) => {
  const data = JSON.parse(event.data);

  // ✅ 결과 확인용 조건별 로그
  if (data.type === "read") {
    console.log(`👁 유저 ${data.user_id}가 메시지 ${data.message_id}를 읽음`);
  } else if (data.type === "online_users") {
    console.log("🟢 현재 접속 중인 유저 ID 목록:", data.user_ids);
  } else if (data.type === "old_messages") {
    console.log("📜 이전 메시지 목록:", data.messages);
  } else {
    console.log("💬 일반 메시지 수신:", data);
  }
};

socket.onclose = () => {
  console.log("❌ WebSocket 연결 종료됨");
};

socket.onerror = (err) => {
  console.error("🚨 WebSocket 에러:", err);
};
